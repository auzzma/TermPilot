cmake_minimum_required(VERSION 3.22)
project(TermPilotMobileMedia LANGUAGES C CXX)
include(../../../scripts/PrivateBuildPaths.cmake)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5 CACHE STRING "Dependency compatibility")
set(BUILD_SHARED_LIBS ON CACHE BOOL "Replaceable codec libraries" FORCE)
set(ENABLE_DECODER OFF CACHE BOOL "No decoder CLI" FORCE)
set(ENABLE_ENCODER OFF CACHE BOOL "No encoder CLI" FORCE)
set(ENABLE_SDL OFF CACHE BOOL "No SDL" FORCE)
set(BUILD_PROGRAMS OFF CACHE BOOL "No FDK tools" FORCE)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/runtime")
add_subdirectory(../../../vendor/libde265 libde265)
add_subdirectory(../../../vendor/fdk-aac fdk-aac)
add_library(termpilot_media SHARED adapter.cpp)
target_include_directories(termpilot_media PRIVATE "${CMAKE_BINARY_DIR}/libde265" ../../../vendor/libde265
 ../../../vendor/fdk-aac/libAACdec/include ../../../vendor/fdk-aac/libSYS/include)
target_link_libraries(termpilot_media PRIVATE de265 fdk-aac)
if(ANDROID)
  target_link_libraries(termpilot_media PRIVATE mediandk)
  target_include_directories(fdk-aac PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/compat")
  target_link_libraries(fdk-aac PRIVATE log)
endif()
