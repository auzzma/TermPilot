cmake_minimum_required(VERSION 3.16)
project(TermPilotMediaCodec LANGUAGES C CXX)
include(../../scripts/PrivateBuildPaths.cmake)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POLICY_VERSION_MINIMUM 3.5 CACHE STRING "Older dependency CMake compatibility")
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS ON CACHE BOOL "Replaceable native codec libraries" FORCE)
set(ENABLE_DECODER OFF CACHE BOOL "No libde265 CLI" FORCE)
set(ENABLE_ENCODER OFF CACHE BOOL "No encoder" FORCE)
set(ENABLE_SDL OFF CACHE BOOL "No SDL" FORCE)
set(BUILD_PROGRAMS OFF CACHE BOOL "No FDK utilities" FORCE)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/runtime")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/runtime")
foreach(configuration DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set(CMAKE_RUNTIME_OUTPUT_DIRECTORY_${configuration} "${CMAKE_BINARY_DIR}/runtime")
  set(CMAKE_LIBRARY_OUTPUT_DIRECTORY_${configuration} "${CMAKE_BINARY_DIR}/runtime")
endforeach()
add_subdirectory(../../vendor/libde265 libde265)
add_subdirectory(../../vendor/fdk-aac fdk-aac)
add_executable(termpilot-media main.cpp)
target_include_directories(termpilot-media PRIVATE "${CMAKE_BINARY_DIR}/libde265" ../../vendor/libde265
  ../../vendor/fdk-aac/libAACdec/include ../../vendor/fdk-aac/libSYS/include)
target_link_libraries(termpilot-media PRIVATE de265 fdk-aac)
if(WIN32)
  target_compile_definitions(termpilot-media PRIVATE NOMINMAX)
  target_link_libraries(termpilot-media PRIVATE mfplat mf mfuuid wmcodecdspuuid ole32 oleaut32)
endif()
if(APPLE)
  set_target_properties(termpilot-media PROPERTIES BUILD_RPATH "@loader_path")
endif()
